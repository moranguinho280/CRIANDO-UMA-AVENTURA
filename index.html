import pandas as pd import matplotlib.pyplot as plt import seaborn as sns # Configuração estética dos gráficos sns.set_theme(style="whitegrid") def executar_analise(caminho_csv): # Carregar a base de dados df = pd.read_csv(caminho_csv) # 1. Filtrar os grupos diario = df[df['frequencia'] == 'Diariamente']['satisfacao'] raro = df[df['frequencia'] == 'Raramente']['satisfacao'] # 2. Criar a visualização (Histograma) plt.figure(figsize=(12, 6)) sns.histplot(diario, kde=True, color="skyblue", label="Uso Diário", element="step") sns.histplot(raro, kde=True, color="orange", label="Uso Raro", element="step") # Customização do gráfico plt.title('Comparação de Satisfação: Usuários Diários vs. Raros', fontsize=15) plt.xlabel('Nota de Satisfação (0-10)', fontsize=12) plt.ylabel('Frequência de Respostas', fontsize=12) plt.legend() # 3. Identificar diferenças (Salvar imagem para o GitHub) plt.savefig('grafico_comparativo.png') plt.show() # 4. Breve comparação estatística print(f"Média Diário: {diario.mean():.2f}") print(f"Média Raro: {raro.mean():.2f}") # Para testar, você precisaria de um arquivo 'transporte.csv' # executar_analise('transporte.csv')
