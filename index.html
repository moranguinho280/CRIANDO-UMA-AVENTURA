import pandas as pd import plotly.express as px import plotly.graph_objects as go # 1. Carregar a base de dados # Supondo um CSV com colunas 'frequencia' e 'satisfacao' df = pd.read_csv('dados_transporte.csv') # 2. Filtrar os grupos de interesse diariamente = df[df['frequencia'] == 'Diariamente']['satisfacao'] raramente = df[df['frequencia'] == 'Raramente']['satisfacao'] # 3. Gerar Histogramas Comparativos fig = go.Figure() fig.add_trace(go.Histogram( x=diariamente, name='Uso Diário', xbins=dict(start=0, end=10, size=1), marker_color='#EB89B5', opacity=0.75 )) fig.add_trace(go.Histogram( x=raramente, name='Uso Raro', xbins=dict(start=0, end=10, size=1), marker_color='#330C73', opacity=0.75 )) # 4. Organizar o layout (Equivalente à "Nova Aba") fig.update_layout( title_text='Distribuição da Satisfação por Frequência de Uso', xaxis_title_text='Nota de Satisfação', yaxis_title_text='Quantidade de Respostas', barmode='overlay', # Sobrepõe as barras para comparação direta template='plotly_white' ) # Salvar o gráfico para subir no GitHub fig.write_image("analise_satisfacao.png") fig.show() # 5. Breve Análise Automatizada print(f"Média Satisfação (Diário): {diariamente.mean():.2f}") print(f"Média Satisfação (Raro): {raramente.mean():.2f}")
